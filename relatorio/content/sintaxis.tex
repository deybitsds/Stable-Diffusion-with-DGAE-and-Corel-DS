% PSEUDOCODIGO

\begin{algorithm}[H]
    \caption{Salvamento do Melhor Modelo (Checkpointing)}
    \label{algo:checkpointing}
    \SetAlgoLined
    \KwIn{Modelo, Carregador de dados de validação}
    \KwOut{Melhor modelo salvo em disco}
    \BlankLine
    Inicializar $best\_val\_loss \leftarrow \infty$\;
    \For{cada época no intervalo de treinamento}{
        Treinar o modelo por uma época\;
        Calcular $avg\_val\_loss$ no conjunto de validação\;
        \If{$avg\_val\_loss < best\_val\_loss$}{
            $best\_val\_loss \leftarrow avg\_val\_loss$\;
            Salvar os pesos do modelo atual em um arquivo\;
        }
    }
\end{algorithm}
